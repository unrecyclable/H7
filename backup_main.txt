int i2c4_reg_receive(uint8_t addr,uint8_t reg,uint8_t*data,int len)
{
	int res=0;
	
	if(HAL_I2C_Master_Transmit(&hi2c4,addr<<1,&reg,1,0xff)!=HAL_OK)
		res=HAL_ERROR;
	if(HAL_I2C_Master_Receive(&hi2c4,addr<<1,data,len,0xff)!=HAL_OK)
		res=HAL_ERROR;

typedef struct{
	float kp;
	float ki;
	float kd;
	float last;
	float integrate;
}PID;

float pid_control(PID*pid,float delta)
{
	float out;
	
	
	pid->integrate+=delta;
	
	if(pid->integrate>=50)
	{
		pid->integrate=50;
	}
	else if(fabs(pid->integrate)<0.01)
		pid->integrate=0;
	if(pid->last*delta<0)
	{
		pid->integrate=0;
	}
	out=delta*pid->kp+(delta-pid->last)*pid->kd+pid->ki*pid->integrate;
	//printf("%f,%f,%f\n",delta*pid->kp,pid->ki*pid->integrate,(delta-pid->last)*pid->kd);
	pid->last=delta;
	if(out>50)
		out=50;
	else if(out<-50)
		out=-50;
	return out;
	
	
}

void serial_parse_data(uint8_t*data,int len)  //0xfe 0x10 xx(0x00~0x08)//??
																							//0xfe 0x20 xx(0x00,0x01)//?
{
	int bufferp=0;
	int flag=0;
	uint8_t ch=0;
	while(bufferp<len)
	{
		ch=data[bufferp];
		switch(flag)
		{
			case 0:
			if(ch==0xfe)
				flag=1;
			break;
			case 1:
				if(ch==0x10)
					flag=2;
				else if(ch==0x20)
					flag=3;
				else if(ch==0x30)
					flag=4;
				else if(ch==0x40)
					flag=5;
				else if(ch==0x50)
					flag=6;
				break;
			case 2:
					move_flag=ch;
					if(data[bufferp+1]!=0)
							move_gain=data[bufferp+1];
					flag=-1;
					break;
			case 3:
					light_flag=ch;
					flag=-1;
					break;
			case 4:
					y_flag=ch;
					if(data[bufferp+1]!=0)
							y_gain=data[bufferp+1];
					flag=-1;
					break;
			case 5:
					cam_flag=ch;
					flag=-1;
					break;
			case 6:
					arm_flag=ch;
					flag=-1;
					break;
			default:
					break;
				
		}
		if(flag==-1)
			break;
		bufferp++;
		}
	
	
}
