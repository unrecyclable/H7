void data_swap(void*data,int len,int width)
{
	uint8_t*p=data;
	for(int i=0;i<len/width;i++)
	{
		for(int j=0;j<width/2;j++)
		{
			int temp=p[i*width+j];
			p[i*width+j]=p[i*width+width-j-1];
			p[i*width+width-j-1]=temp;
		}
	}
	
}
void serial_init()
{
	MX_USART1_UART_Init();
	// 启动DMA接收
    HAL_UART_Receive_DMA(&huart1, serial_rx_buffer, serial_rx_buffer_size);
    // 使能空闲中断
    __HAL_UART_ENABLE_IT(&huart1, UART_IT_IDLE);
}
